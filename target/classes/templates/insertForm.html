Modèles de flux RSS25SB

Cliquez sur un bouton pour charger un modèle :

Modèle minimal

<?xml version="1.0" encoding="UTF-8"?>
<feed xmlns="http://www.w3.org/2005/Atom" lang="fr" version="25">
    <title>Titre du flux</title>
    <pubDate>2025-05-19T10:00:00Z</pubDate>
    <copyright>Copyright 2025</copyright>
    <link rel="self" type="application/atom+xml" href="https://example.com/feed"/>
    <item>
        <guid>https://example.com/article/1</guid>
        <title>Premier article</title>
        <category term="actualité"/>
        <published>2025-05-18T09:00:00Z</published>
        <content type="text">Contenu de l'article</content>
        <author>
            <name>Nom de l'auteur</name>
        </author>
    </item>
</feed>
